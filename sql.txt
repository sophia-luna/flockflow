create table public.galpoes (
  id uuid not null default gen_random_uuid(),
  nome character varying(255) not null,
  custo_implantacao_iot numeric(10, 2) null default 0.00,
  constraint galpoes_pkey primary key (id)
) TABLESPACE pg_default;

create table public.lotes_avicolas (
  id uuid not null default gen_random_uuid(),
  galpao_id uuid not null,
  data_alojamento timestamp with time zone not null default now(),
  data_saida timestamp with time zone null, 
  quantidade_aves integer not null default 15000,
  ativo boolean null default true,
  created_at timestamp with time zone null default now(),
  constraint lotes_avicolas_pkey primary key (id),
  constraint lotes_avicolas_galpao_id_fkey foreign key (galpao_id) references public.galpoes (id) on delete restrict
) TABLESPACE pg_default;

create table public.configuracoes_operacionais (
  id uuid not null default gen_random_uuid(),
  galpao_id uuid not null,
  potencia_exaustor_kw numeric not null,
  tarifa_kwh numeric not null,
  custo_fixo_ciclo numeric not null,
  receita_projetada numeric not null,
  atualizado_em timestamp with time zone null default timezone ('utc'::text, now()),
  constraint configuracoes_operacionais_pkey primary key (id),
  constraint config_operacionais_galpao_id_fkey foreign key (galpao_id) references public.galpoes (id) on delete cascade
) TABLESPACE pg_default;

create table public.telemetria_avicola (
  id bigint generated by default as identity not null,
  timestamp timestamp with time zone not null default now(),
  galpao_id uuid null,
  temperatura_c numeric(5, 2) null,
  umidade_perc numeric(5, 2) null,
  velocidade_ar_ms numeric(5, 2) null,
  nh3_ppm numeric(5, 2) null,
  co2_ppm integer null,
  poeira_pm25 numeric null,
  created_at timestamp with time zone null default now(),
  constraint telemetria_avicola_pkey primary key (id),
  constraint telemetria_avicola_galpao_id_fkey foreign key (galpao_id) references public.galpoes (id) on delete cascade
) TABLESPACE pg_default;

create index idx_telemetria_tempo on public.telemetria_avicola using btree ("timestamp" desc, galpao_id) TABLESPACE pg_default;
